cmake_minimum_required(VERSION 3.25)
project(compiler_output_contract LANGUAGES CXX)
set(PROJECT_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../..")
include("${PROJECT_SOURCE_DIR}/cmake/InfernuxCompilerOutput.cmake")
add_library(compiler_contract SHARED dummy.cpp)
infernux_compiler_output(compiler_contract)
foreach(property RUNTIME_OUTPUT_DIRECTORY LIBRARY_OUTPUT_DIRECTORY)
    get_target_property(output compiler_contract ${property})
    if(NOT output STREQUAL "${PROJECT_BINARY_DIR}/wheel/Infernux/_compiler/taichi/_vendor/taichi/_lib/core/$<0:>")
        message(FATAL_ERROR "${property} does not target the private engine compiler: ${output}")
    endif()
endforeach()
